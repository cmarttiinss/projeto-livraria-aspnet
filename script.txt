/* CRIANDO OS DATABASES PARA UMA LIVRARIA */
CREATE DATABASE db_livraria;
/*--------------------------*/

/* VERIFICANDO SE O AMBIENTE LIVRARIA FOI CRIADO */ 
SHOW DATABASES;

/*SELECIONANDO BANCO DE DADOS PARA USO */
USE db_livraria;

/* CRIANDO TABELAS PARA A LIVRARIA */

/* CRIANDO TABELA LOGIN */
CREATE TABLE tb_login (
	id INT PRIMARY KEY AUTO_INCREMENT,
    nome VARCHAR(120) NOT NULL,
    email VARCHAR(120) UNIQUE NOT NULL,
    senha VARCHAR(120),
    senhahash VARCHAR(120) NOT NULL 
);

/* CONFERINDO O BANCO E SUAS TABELAS  */
SHOW DATABASES;
SHOW TABLES;

/* CRIANDO TABELA LIVRO */
CREATE TABLE tb_livro (
	id INT PRIMARY KEY AUTO_INCREMENT,
    isbn CHAR(20) NOT NULL UNIQUE,
    nome VARCHAR(120) NOT NULL,
    descricao TEXT,
    valor DECIMAL(18,2) NOT NULL
);

/* CONFERINDO O BANCO E SUAS TABELAS */
SHOW DATABASES;
SHOW TABLES;
